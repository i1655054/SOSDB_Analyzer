Procedure: UF_Calendar5.GP_MakeCalendar  [ カレンダー生成＜UF_Calendar5＞ ]
HCP	Business
┬
〇│Private Sub GP_MakeCalendar()	カレンダー生成＜UF_Calendar5＞
┴└┐
　　□ Dim dteDate As Date                                             ' 日付WORK
　　□ Dim intYobi As Integer                                          ' 曜日位置INDEX
　　□ Dim intYear As Integer                                          ' 指定年
　　□ Dim lngCurStrIx As Long                                         ' 当月開始INDEX
　　□ Dim lngCurEndIx As Long                                         ' 当月終了INDEX
　　□ Dim lngIx As Long                                               ' フォーム上位置INDEX
　　□ Dim lngIx2 As Long                                              ' テーブルINDEX(Work)
　　□ Dim lngIxC As Long                                              ' カレンダーINDEX
　　□ Dim lngIxCMax As Long                                           ' カレンダーINDEX上限
　　□ Dim lngCurPos As Long                                           ' 選択日の位置
　　□ Dim tblCalendar() As g_typAboutCalendar2                        ' カレンダーテーブル
　　〇 intYear = Year(g_FormDate1)                                 ' 指定年
　　│#  表示範囲確認
　　◆┐If ((intYear < g_cnsYearFrom) Or  (intYear > (Year(Date) + g_cnsYearToAdd))) Then
　　│〇 MsgBox "祝日計算範囲を超えています。", vbExclamation, Me.Caption
　　│〇 g_FormDate1 = g_tblDateLabel(g_CurPos).Hiduke
　　│┴End If
　　│#  表示年月設定
　　〇 g_intCurYear = Year(g_FormDate1)                            ' 指定年
　　〇 g_intCurMonth = Month(g_FormDate1)                          ' 指定月
　　〇 LBL_YM.Caption = g_intCurYear & "年" & Format(g_intCurMonth, "00") & "月"
　　│#  カレンダーテーブル生成
　　◎ Call modAboutCalendar2.GP_GetCalendarTable3(g_intCurYear,  g_intCurMonth,  tblCalendar,  lngCurStrIx,  lngCurEndIx)	三か月カレンダー生成＜modAboutCalendar2＞
　　│#  カレンダー開始位置算出
　　〇 lngIxCMax = UBound(tblCalendar)
　　〇 lngIxC = lngCurStrIx
　　〇 lngIxC = lngIxC - tblCalendar(lngCurStrIx).Yobi
　　◆┐If g_cnsStartYobi = 2 Then
　　│〇 lngIxC = lngIxC + 1
　　│◆ If lngIxC > lngCurStrIx Then lngIxC = lngIxC - 7
　　│┴End If
　　〇 intYobi = 0
　　〇 lngCurPos = -1
　　│#  日付ラベル生成
　　＠┐For lngIx = 0 To 41
　　││#  日付情報設定
　　│〇 intYobi = intYobi + 1
　　│◆ If intYobi > 7 Then intYobi = 1
　　│〇 dteDate = tblCalendar(lngIxC).Hiduke
　　│◆┐If dteDate = g_FormDate1 Then
　　││〇 lngCurPos = lngIx
　　││┴End If
　　│＠┐With g_tblDateLabel(lngIx)
　　│││#  ガイド情報設定
　　││〇 .Hiduke = dteDate
　　││〇 .Yobi = tblCalendar(lngIxC).Yobi
　　││〇 .Syuku = tblCalendar(lngIxC).Syuku
　　││〇 .StsGuide = Format(dteDate, g_cnsDateFormat) & g_tblYobi(.Yobi)
　　││◆┐If .Syuku <> 0 Then
　　│││〇 .StsGuide = .StsGuide & " " & tblCalendar(lngIxC).SyukuNm
　　│││┴End If
　　││┴End With
　　││#  月初月末位置取得
　　│◆┐If lngIxC = lngCurStrIx Then
　　││〇 g_CurPosF = lngIx
　　│├┐ElseIf lngIxC = lngCurEndIx Then
　　││〇 g_CurPosT = lngIx
　　││┴End If
　　│〇 g_tblDateLabel(lngIx).Label.Caption = Day(dteDate)
　　││#  表示色設定
　　│◎ Call GP_SetForeColor(lngIx, g_FormDate1)	カレンダー表示色設定＜UF_Calendar5＞
　　│〇 lngIxC = lngIxC + 1
　　│┴Next lngIx
　　│#  ガイド更新
　　〇 LBL_GUIDE.Caption = g_tblDateLabel(g_CurPos).StsGuide   ' ガイド表示
　　┴End Sub
