〇│Private Sub GP_SetForeColor(ByVal lngPos As Long, ByVal dteCurDate As Date)カレンダー表示色設定＜UF_Calendar5＞
┴└┐
　　□ Dim dteDate As Date ' 日付Work
　　□ Dim lngYear As Long ' 現在年
　　□ Dim lngMonth As Long ' 現在月
　　│# 表示属性初期化
　　＠┐With g_tblDateLabel(lngPos)
　　│〇 dteDate = .Hiduke
　　│〇 lngYear = Year(dteDate)
　　│〇 lngMonth = Month(dteDate)
　　│〇 .Label.Font.Bold = False
　　│〇 .Label.ForeColor = cnsFC_Normal
　　││# 選択日判定
　　│◆┐If dteDate = dteCurDate Then
　　││〇 .Label.BackColor = cnsBC_Select
　　│││# 選択日表示
　　││〇 g_CurPos = lngPos
　　│├┐ElseIf ((lngYear = g_intCurYear) And (lngMonth = g_intCurMonth)) Then ' 当月内か
　　│││# 当月日付表示
　　││◆┐Select Case .Yobi
　　│││├┐Case 0 ' 日曜日
　　││││〇 .Label.BackColor = cnsBC_Sunday
　　│││├┐Case 6 ' 土曜日
　　││││〇 .Label.BackColor = cnsBC_Saturday
　　│││├┐Case Else
　　││││〇 .Label.BackColor = cnsBC_Month
　　│││┴End Select
　　│├┐ELSE
　　│││# 他月日付表示
　　││〇 .Label.BackColor = cnsBC_Other
　　││┴End If
　　││# 祝日表示設定
　　│◆┐If .Syuku <> 0 Then
　　││〇 .Label.ForeColor = cnsFC_Hori
　　││◆ If ((lngYear = g_intCurYear) And (lngMonth = g_intCurMonth)) Then .Label.Font.Bold = True
　　││┴End If
　　│┴End With
　　┴End Sub
