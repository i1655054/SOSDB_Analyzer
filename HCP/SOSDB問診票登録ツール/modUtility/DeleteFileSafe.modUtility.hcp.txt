〇│Public Sub DeleteFileSafe(ByVal filePath As String)	安全ファイル削除＜modUtility＞
┴└┐
　　☆ On Error Resume Next
　　│#  ファイル存在確認
　　◆┐If Trim(filePath) <> "" And Dir(filePath) <> "" Then
　　││#  ファイル削除
　　│〇 Kill filePath
　　││#  削除結果確認

　　││〇 Debug.Print "削除失敗: " & filePath & " / " & Err.Description
　　│├┐ELSE
　　││┴End If
　　│┴End If
　　☆ On Error GoTo 0
　　┴End Sub
