Procedure: clsAboutCalendar2.FP_GetHoliParameter  [ 祝日パラメータ読込＜clsAboutCalendar2＞ ]
HCP	Business
┬
〇│Private Function FP_GetHoliParameter(Optional ByVal blnOmitCheck As Boolean = False) As Boolean	祝日パラメータ読込＜clsAboutCalendar2＞
┴└┐
　　□ Dim lngIx As Long                                               ' テーブルINDEX
　　□ Dim lngRow As Long                                              ' 行INDEX
　　□ Dim lngRowMax As Long                                           ' 行INDEX上限
　　□ Dim lngCurrM As Long                                            ' 現在月
　　□ Dim lngPrevM As Long                                            ' 直前月
　　□ Dim objParaSheet As Worksheet                                   ' 祝日パラメータシート
　　〇 FP_GetHoliParameter = False
　　│#  状態初期化
　　〇 g_blnParaReaded = False
　　〇 g_blnParaChecked = False
　　☆ On Error Resume Next
　　│#  パラメータシート取得
　　〇 Set objParaSheet = ThisWorkbook.Worksheets(g_cnsParaSheet)
　　│#  シート存在確認
　　◆┐If Err.Number <> 0 Then
　　│〇 MsgBox "「" & g_cnsParaSheet & "」シートが参照できません。", vbCritical, g_cnsTitle
　　│☆ On Error GoTo 0
　　│→ Exit Function
　　│┴End If
　　☆ On Error GoTo 0
　　│#  パラメータ検証
　　◆┐If Not g_blnParaChecked And Not blnOmitCheck Then
　　│◆→ If Not FP_CheckSyukuParameter(objParaSheet, lngRowMax) Then Exit Function
　　├┐ELSE
　　│＠┐With objParaSheet
　　││┴End With
　　│〇 lngRowMax = .Range("$A$" & .Rows.count).End(xlUp).row
　　│┴End If
　　│#  テーブル領域確保
　　〇 ReDim g_tblParamater(lngRowMax - 3)
　　│#  月別位置テーブル初期化
　　＠┐For lngIx = 1 To 12
　　│＠┐With g_tblParamMMPos(lngIx)
　　││〇 .StrIx = -1
　　││〇 .EndIx = -1
　　││┴End With
　　│┴Next lngIx
　　〇 lngIx = 0
　　〇 lngRow = 3
　　＠┐With objParaSheet                                                 ' ｢祝日パラメータ｣シート
　　│〇 lngCurrM = .Cells(lngRow, 1).Value
　　││#  パラメータ読込
　　│＠┐Do While lngRow <= lngRowMax
　　│││#  月開始位置設定
　　││〇 lngPrevM = lngCurrM
　　││〇 g_tblParamMMPos(lngPrevM).StrIx = lngIx
　　│││#  月内パラメータ読込
　　││＠┐Do While lngCurrM = lngPrevM
　　│││＠┐With g_tblParamater(lngIx)
　　││││〇 .Getsu = lngPrevM                               ' 月
　　││││◎ .SyoriKbn = FP_GetNumber(objParaSheet.Cells(lngRow, 2))     ' 処理区分	数値取得＜clsAboutCalendar2＞
　　│││││	内部呼出:Cells
　　││││◎ .Hiduke = FP_GetNumber(objParaSheet.Cells(lngRow, 3))       ' [固定日]日	数値取得＜clsAboutCalendar2＞
　　│││││	内部呼出:Cells
　　││││◎ .FurikaeKbn = FP_GetNumber(objParaSheet.Cells(lngRow, 4))   ' [固定日]振替区分	数値取得＜clsAboutCalendar2＞
　　│││││	内部呼出:Cells
　　││││◎ .HmSyusu = FP_GetNumber(objParaSheet.Cells(lngRow, 5))      ' [HM]週数	数値取得＜clsAboutCalendar2＞
　　│││││	内部呼出:Cells
　　││││◎ .HmYobi = FP_GetNumber(objParaSheet.Cells(lngRow, 6))       ' [HM]曜日	数値取得＜clsAboutCalendar2＞
　　│││││	内部呼出:Cells
　　││││〇 .SyukuNm = Trim(objParaSheet.Cells(lngRow, 7).Value)        ' 祝日名
　　││││◎ .StrYear = FP_GetNumber(objParaSheet.Cells(lngRow, 8))      ' 開始年	数値取得＜clsAboutCalendar2＞
　　│││││	内部呼出:Cells
　　││││◎ .EndYear = FP_GetNumber(objParaSheet.Cells(lngRow, 9))      ' 終了年	数値取得＜clsAboutCalendar2＞
　　│││││	内部呼出:Cells
　　││││┴End With
　　│││〇 lngIx = lngIx + 1
　　│││〇 lngRow = lngRow + 1
　　│││◆┐If lngRow <= lngRowMax Then
　　││││〇 lngCurrM = .Cells(lngRow, 1).Value
　　│││├┐ELSE
　　││││〇 lngCurrM = 99
　　││││┴End If
　　│││┴Loop
　　│││#  月終了位置設定
　　││〇 g_tblParamMMPos(lngPrevM).EndIx = lngIx - 1
　　││┴Loop
　　│┴End With
　　│#  読込完了
　　〇 g_blnParaReaded = True
　　〇 g_blnParaChecked = True
　　〇 FP_GetHoliParameter = True
　　┴End Function
