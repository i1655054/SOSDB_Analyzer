Option Explicit

' @JPName
' HCP記号取得
'
' @Category
' HCP
'
' @Input
' Node(clsHCPNode)
'
' @Output
' Symbol(String)
'
' @Summary
' ノード種別に対応する
' HCP表示記号を取得する
'
' @Remarks
' IF、CALL、FORなどの
' ノード種別に応じた記号を返却する
'
Public Function GetHCPSymbol( _
                ByVal Node As clsHCPNode) _
                As String

    '# HCP記号判定
    Select Case Node.NodeType

        Case NODE_START
            GetHCPSymbol = "〇│"

        Case NODE_DECLARE
            GetHCPSymbol = "□ "

        Case NODE_REFERENCE
            GetHCPSymbol = "◇ "

        Case NODE_CONTROL
            GetHCPSymbol = "☆ "

        Case NODE_NORMAL
            GetHCPSymbol = "〇 "

        Case NODE_CALL
            GetHCPSymbol = "◎ "

        Case NODE_IF
            GetHCPSymbol = "◆ "
        
        Case NODE_INLINE_IF
        
            If InStr(1, _
                Node.Text, _
                "Exit ", _
                vbTextCompare) > 0 Then

                GetHCPSymbol = "◆→ "

            Else

                GetHCPSymbol = "◆ "

            End If

        Case NODE_TRUE
            GetHCPSymbol = "THEN "

        Case NODE_FALSE
            GetHCPSymbol = "ELSE "

        Case NODE_FOR
            GetHCPSymbol = "＠ "
        
        Case NODE_DO
            GetHCPSymbol = "＠ "

        Case NODE_WITH
            GetHCPSymbol = "＠ "

        Case NODE_WHILE
            GetHCPSymbol = "＠ "

        Case NODE_SELECT
            GetHCPSymbol = "◆ "

        Case NODE_CASE
            GetHCPSymbol = ""

        Case NODE_MERGE
            GetHCPSymbol = ""

        Case NODE_END
            GetHCPSymbol = "┴"
        
        Case NODE_GOTO
            GetHCPSymbol = "→ "
    
        Case NODE_EXIT
            GetHCPSymbol = "→ "
        
        Case NODE_LABEL
            GetHCPSymbol = "▼ "
        
        Case NODE_RESUME
            GetHCPSymbol = "☆ "

        Case Else
            GetHCPSymbol = ""

    End Select

End Function

'===================================================================
' @JPName
' Business継続プレフィックス取得
'
' @Category
' HTML
'
' @Input
' RenderText(String)
'
' @Output
' Prefix(String)
'
' @Summary
' Business複数行出力時の
' HCP継続行プレフィックスを取得する
'
' @Remarks
' HCP表示記号位置を基準に
' 継続行表示用の
' 「││」プレフィックスを生成する
'
'===================================================================
Public Function GetBusinessPrefix( _
                ByVal RenderText As String) _
                As String

    Dim Pos As Long

    '# CALL記号検索
    Pos = InStr( _
                RenderText, _
                "◎")

    If Pos > 0 Then

        GetBusinessPrefix = _
            Left$( _
                RenderText, _
                Pos - 1) & _
            "│"

        Exit Function

    End If

    '# REFERENCE記号検索
    Pos = InStr( _
                RenderText, _
                "◇")

    If Pos > 0 Then

        GetBusinessPrefix = _
            Left$( _
                RenderText, _
                Pos - 1) & _
            "│"

        Exit Function

    End If

    '# NORMAL記号検索
    Pos = InStr( _
                RenderText, _
                "〇")

    If Pos > 0 Then

        GetBusinessPrefix = _
            Left$( _
                RenderText, _
                Pos - 1) & _
            "│"

        Exit Function

    End If

    '# DECLARE記号検索
    Pos = InStr( _
                RenderText, _
                "□")

    If Pos > 0 Then

        GetBusinessPrefix = _
            Left$( _
                RenderText, _
                Pos - 1) & _
            "│"

        Exit Function

    End If

    '# IF記号検索
    Pos = InStr( _
                RenderText, _
                "◆")

    If Pos > 0 Then

        GetBusinessPrefix = _
            Left$( _
                RenderText, _
                Pos - 1) & _
            "│"

        Exit Function

    End If

    '# FOR記号検索
    Pos = InStr( _
                RenderText, _
                "＠")

    If Pos > 0 Then

        GetBusinessPrefix = _
            Left$( _
                RenderText, _
                Pos - 1) & _
            "│"

        Exit Function

    End If

    '# 継続行生成不可
    GetBusinessPrefix = "│"

End Function

'===================================================================
' @JPName
' Procedureリンク生成
'
' @Category
' HCP
'
' @Input
' ProcName : Procedure名
'
' @Output
' HtmlText
'
' @Summary
' Procedure名に対応する
' HTMLリンクを生成する
'
' @Remarks
' Procedure名から
' HCP HTML相対パスを取得し
'
' <a href="...">
' </a>
'
' を生成して返却する
'
' Procedure未登録時は
' Procedure名をそのまま返却する
'===================================================================
Public Function BuildProcedureLink( _
                ByVal ProcName As String) _
                As String

    '# HTMLパス
    Dim HtmlPath As String

    '# Procedure名確認
    If Trim$(ProcName) = "" Then

        BuildProcedureLink = ""

        Exit Function

    End If

    '# Procedure HTMLパス取得
    HtmlPath = _
        GetProcedureHtmlPath( _
            ProcName)

    '# Procedure未登録
    If HtmlPath = "" Then

        BuildProcedureLink = _
            ProcName

        Exit Function

    End If

    '# Procedureリンク生成
    BuildProcedureLink = _
        "<a href=""" & _
        HtmlPath & _
        """>" & _
        ProcName & _
        "</a>"

End Function

'===================================================================
' @JPName
' ProcedureHTMLパス取得
'
' @Category
' HCP
'
' @Input
' ProcName : Procedure名
'
' @Output
' HtmlPath
'
' @Summary
' Procedureに対応する
' HCP HTML相対パスを返却する
'
' @Remarks
' ProcListからModule名を取得し
'
' ../ModuleName/
' ProcName.ModuleName.hcp.html
'
' 形式の相対パスを返却する
'
' Procedure未登録時は
' 空文字列を返却する
'
'===================================================================
Public Function GetProcedureHtmlPath( _
                ByVal ProcName As String) _
                As String

    '# ProcList
    Dim ws As Worksheet

    '# 最終行
    Dim LastRow As Long

    '# 行番号
    Dim r As Long

    '# Module名
    Dim ModuleName As String

    Set ws = _
        gContext.ManagementBook.Worksheets( _
            SHEET_PROC_LIST)

    LastRow = _
        ws.Cells( _
            ws.Rows.Count, _
            COL_MODULE).End(xlUp).row

    For r = 2 To LastRow

        If StrComp( _
                Trim$(ws.Cells( _
                    r, _
                    COL_PROC).Value), _
                ProcName, _
                vbTextCompare) = 0 Then

            ModuleName = _
                Trim$(ws.Cells( _
                    r, _
                    COL_MODULE).Value)

            GetProcedureHtmlPath = _
                "../" & _
                ModuleName & "/" & _
                ProcName & "." & _
                ModuleName & _
                ".hcp.html"

            Exit Function

        End If

    Next r

End Function
