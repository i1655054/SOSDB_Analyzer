
'===================================================================
'@JPName     問診情報取得
'@Category   問診票
'@Input      xlBook(Object);sheetType(enMonshinSheetType)
'@Output     MonshinInfoType
'@Summary    問診票から情報を取得してMonshinInfo構造体を生成する
'@Remarks    MAP定義を利用してセル値を取得し説明文および備考を生成する
'===================================================================
' 問診票情報取得（メイン関数）
Public Function Get_MonshinInfo( _
    ByVal xlBook As Object, _
    ByVal sheetType As enMonshinSheetType _
) As MonshinInfoType

    '# 変数初期化
    Dim info As MonshinInfoType
    Dim Key As Variant
    Dim addr As String
    Dim Value As String
    Dim wsName As String

    '# MAP初期化確認
    If modMonshinMap.MonshinMap Is Nothing Then
        modMonshinMap.InitMonshinCommon
    End If

    '# 問診情報初期化
    info = InitMonshinInfo(sheetType)
    
    '# 対象シート名取得
    wsName = modMonshinMap.MonShinSheet_id(sheetType)
    
    '# MAP定義に従い問診情報取得
    For Each Key In modMonshinMap.MonshinMap.Keys

        addr = modMonshinMap.MonshinMap(Key)(sheetType)

        '# セル値取得
        If addr <> "" Then
            Value = GetValueByAddress(xlBook, wsName, addr)
        Else
            Value = ""
        End If
        
        '# 問診情報設定
        SetMonshinValue info, CStr(Key), Value
    Next Key
    
    '# 依頼元管理番号生成
    info.RequesterNum = BuildRequesterNum(info)
    
    '# 説明文生成
    info.Description = BuildDescription(info)
    
    '# 備考生成
    info.Remarks = BuildRemarks(info)

    '# 問診情報返却
    Get_MonshinInfo = info

End Function
